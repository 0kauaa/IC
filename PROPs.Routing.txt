{-# LANGUAGE DataKinds, ExplicitForAll #-}

module Sandbox.PROPs.Routing
    ( copy
    , delete
    , swap
    , merge 
    ) where

import Core.PROPsLearner (PROPsLearner(..))
import Core.Multi        (Multi(..))
import Core.Params       (Params(..), type (++))
import Core.Utils        (appendMulti)

copy :: PROPsLearner '[] as (as ++ as)
copy = PROPsLearner
    { iP        = \ParamsNull as   -> appendMulti as as
    , uP        = \ParamsNull _ _  -> ParamsNull
    , rP        = \ParamsNull _ gs ->
        let n        = multiLen gs `div` 2
            (g1, g2) = splitMulti n gs
        in  zipMulti (+) g1 g2
    , iniParamsP = ParamsNull
    }

delete :: PROPsLearner '[] as '[]
delete = PROPsLearner
    { iP         = \ParamsNull _    -> MultiNull
    , uP         = \ParamsNull _ _  -> ParamsNull
    , rP         = \ParamsNull as _ -> zeroMulti as
    , iniParamsP = ParamsNull
    }

swap :: PROPsLearner '[] (as ++ bs) (bs ++ as)
swap = PROPsLearner
    { iP = \ParamsNull entry ->
        let n        = multiLen entry `div` 2
            (as, bs) = splitMulti n entry
        in  appendMulti bs as
    , uP = \ParamsNull _ _   -> ParamsNull
    , rP = \ParamsNull _ gs  ->
        let n        = multiLen gs `div` 2
            (bs, as) = splitMulti n gs
        in  appendMulti as bs
    , iniParamsP = ParamsNull
    }

merge :: Num a => PROPsLearner '[] (as ++ as) as
merge = PROPsLearner
    { iP = \ParamsNull entry ->
        let n        = multiLen entry `div` 2
            (as, bs) = splitMulti n entry
        in  zipMulti (+) as bs
    , uP = \ParamsNull _ _   -> ParamsNull
    , rP = \ParamsNull _ gs  -> appendMulti gs gs
    , iniParamsP = ParamsNull
    }


-- funções auxiliares
zipMulti :: (forall a. Num a => a -> a -> a) -> Multi as -> Multi as -> Multi as
zipMulti _ MultiNull MultiNull   = MultiNull
zipMulti f (x :-: xs) (y :-: ys) = f x y :-: zipMulti f xs ys

zeroMulti :: Num a => Multi as -> Multi as
zeroMulti MultiNull  = MultiNull
zeroMulti (_ :-: xs) = 0 :-: zeroMulti xs
